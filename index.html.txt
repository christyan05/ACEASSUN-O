    <script>
        // Configuração inicial das abas e menu mobile
        document.addEventListener('DOMContentLoaded', () => {
            switchTab('edital'); // Garante que a Home (Raio-X) abra primeiro
        });

        // Função para alternar Sidebar em telas pequenas (Mobile)
        function toggleSidebar() {
            const sidebar = document.getElementById('sidebar');
            const overlay = document.getElementById('sidebarOverlay');
            
            // Toggle translate-x
            if(sidebar.classList.contains('-translate-x-full')) {
                sidebar.classList.remove('-translate-x-full');
                overlay.classList.remove('hidden');
            } else {
                sidebar.classList.add('-translate-x-full');
                overlay.classList.add('hidden');
            }
        }

        // Função para alternar as abas do sistema
        function switchTab(tabId) {
            const tabs = document.querySelectorAll('.tab-section');
            tabs.forEach(tab => tab.classList.add('hidden'));

            const navBtns = document.querySelectorAll('.nav-btn');
            navBtns.forEach(btn => {
                btn.classList.remove('bg-brand-600', 'text-white');
                btn.classList.add('text-slate-300', 'hover:bg-slate-800', 'hover:text-white');
            });

            document.getElementById('tab-' + tabId).classList.remove('hidden');

            // Atualiza botões da sidebar se existir o botão
            const activeBtn = document.getElementById('btn-' + tabId);
            if(activeBtn) {
                activeBtn.classList.remove('text-slate-300', 'hover:bg-slate-800', 'hover:text-white');
                activeBtn.classList.add('bg-brand-600', 'text-white');
            }

            // Atualiza o Title no topo
            const titleElement = document.getElementById('page-title');
            if (tabId === 'dashboard') titleElement.innerText = "Cronograma de Estudos (Semana 1)";
            if (tabId === 'edital') titleElement.innerText = "Raio-X do Edital";
            if (tabId === 'materiais') titleElement.innerText = "Módulo de Aulas";
            if (tabId === 'aula1') titleElement.innerText = "Aula 01 - SUS na CF/88";

            // Fechar sidebar no mobile ao clicar em um link
            if(window.innerWidth < 768 && tabId !== 'aula1') {
                const sidebar = document.getElementById('sidebar');
                if(!sidebar.classList.contains('-translate-x-full')) {
                    toggleSidebar();
                }
            }
            
            // Scroll to top
            document.querySelector('main').scrollTo(0,0);
        }

        function msgDisponivel() {
            mostrarMensagem("Aulas em Geração", "O Mentor está compilando a próxima aula em formato interativo. Por enquanto, explore a 'Aula 01' na segunda-feira para ver o padrão!");
        }

        function mostrarMensagem(titulo, texto) {
            const modalAntigo = document.getElementById('custom-modal');
            if(modalAntigo) modalAntigo.remove();

            const modalHTML = `
                <div id="custom-modal" class="fixed inset-0 z-50 flex items-center justify-center p-4 bg-slate-900/60 backdrop-blur-sm">
                    <div class="bg-white rounded-2xl shadow-2xl p-6 max-w-md w-full border border-slate-200 transform transition-all">
                        <div class="flex items-center justify-between mb-4">
                            <h3 class="text-xl font-bold text-slate-800 flex items-center gap-2">
                                <i class="fa-solid fa-circle-info text-brand-500"></i> ${titulo}
                            </h3>
                            <button onclick="document.getElementById('custom-modal').remove()" class="text-slate-400 hover:text-slate-700">
                                <i class="fa-solid fa-xmark text-lg"></i>
                            </button>
                        </div>
                        <p class="text-slate-600 mb-6">${texto}</p>
                        <div class="text-right">
                            <button onclick="document.getElementById('custom-modal').remove()" class="bg-slate-800 text-white font-bold py-2 px-6 rounded-lg hover:bg-slate-700 transition-colors">
                                Entendido
                            </button>
                        </div>
                    </div>
                </div>
            `;
            document.body.insertAdjacentHTML('beforeend', modalHTML);
        }
